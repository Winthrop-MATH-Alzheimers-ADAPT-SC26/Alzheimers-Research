\begin{algorithm}[]
\caption{Forward--Backward Sweep Method}
\label{alg:FBSM}
\begin{algorithmic}[1]

\Require Initial state $x_0$, final time $T$, control bounds
$u_i^{\max}$, weights $w_i$, tolerance $\varepsilon$,
maximum iterations $N_{\max}$

\State Discretize $[0,T]$ into grid points $\{t_k\}_{k=0}^{n}$

\State Compute baseline solution by solving the state system with
$u_1(t)=u_2(t)=u_3(t)=0$

\State Initialize controls:
\[
u_1^{(0)}(t_k)=u_2^{(0)}(t_k)=u_3^{(0)}(t_k)=0
\]

\For{$m = 0,1,\ldots,N_{\max}$}

    \State Solve the state equations forward in time:
    \[
    \dot{x}=f(x,u_1^{(m)},u_2^{(m)},u_3^{(m)}),
    \qquad
    x(0)=x_0
    \]

    \State Solve the adjoint equations backward in time:
    \[
    \dot{\lambda}
    =
    -\frac{\partial H}{\partial x},
    \qquad
    \lambda(T)=0
    \]

    \For{each grid point $t_k$}

        \State Update controls using the optimality conditions
        \[
        \tilde u_1(t_k)
        =
        \min\!\left\{
        u_1^{\max},
        \max\!\left(
        0,
        \frac{\lambda_1(t_k)A_\beta(t_k)}
        {2w_3}
        \right)
        \right\},
        \]

        \[
        \tilde u_2(t_k)
        =
        \min\!\left\{
        u_2^{\max},
        \max\!\left(
        0,
        \frac{\lambda_2(t_k)Ca(t_k)}
        {2w_4}
        \right)
        \right\},
        \]

        \[
        \tilde u_3(t_k)
        =
        \min\!\left\{
        u_3^{\max},
        \max\!\left(
        0,
        \frac{\lambda_3(t_k)\tau(t_k)}
        {2w_5}
        \right)
        \right\}
        \]

    \EndFor

    \State Enforce
    \[
    \tilde u_1(0)=\tilde u_2(0)=\tilde u_3(0)=0
    \]

    \State Apply relaxation:
    \[
    u_i^{(m+1)}
    =
    \frac{1}{2}\tilde u_i
    +
    \frac{1}{2}u_i^{(m)},
    \qquad i=1,2,3
    \]

    \State Compute relative error
    \[
    E
    =
    \max_{i=1,2,3}
    \frac{\|u_i^{(m+1)}-u_i^{(m)}\|}
         {\|u_i^{(m)}\|+10^{-10}}
    \]

    \If{$E < \varepsilon$}
        \State \textbf{break}
    \EndIf

\EndFor

\State Solve the state and adjoint systems one final time using the
converged controls.

\State \Return optimal state trajectory $x^*(t)$, adjoint trajectory
$\lambda^*(t)$, and controls
$\left(u_1^*(t),u_2^*(t),u_3^*(t)\right)$

\end{algorithmic}
\end{algorithm}
